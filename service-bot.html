Discord Bot with Python

import discord
from discord.ext import commands

intents = discord.Intents.all()
bot = commands.Bot(command_prefix='!', intents=intents)

@bot.tree.command(name="ban", description="Ban a member")
@discord.app_commands.describe(member="Member to ban", reason="Reason")
async def ban(interaction: discord.Interaction, member: discord.Member, reason: str = "No reason"):
    if not interaction.user.guild_permissions.ban_members:
        await interaction.response.send_message("No permission!", ephemeral=True)
        return
    await member.ban(reason=reason)
    await interaction.response.send_message(f"Banned {member.mention}: {reason}")

@bot.event
async def on_ready():
    await bot.tree.sync()
    print(f'{bot.user} is ready!')

Telegram Bot

from telethon import TelegramClient, events

api_id = 12345
api_hash = 'your_api_hash'
bot = TelegramClient('bot', api_id, api_hash).start(bot_token='your_bot_token')

@bot.on(events.NewMessage(pattern='/start'))
async def start(event):
    await event.respond('Welcome! I am your bot.')

@bot.on(events.NewMessage(pattern='/help'))
async def help(event):
    await event.respond('Available commands:\n/start - Start bot\n/help - Show help')

bot.run_until_disconnected()

Step 4: Database Integration

For persistent data like user profiles, settings, and logs, I use MongoDB or PostgreSQL:

import sqlite3

# Create database connection
db = sqlite3.connect('bot_database.db')
cursor = db.cursor()

# Users table
cursor.execute('''
    CREATE TABLE IF NOT EXISTS users (
        user_id INTEGER PRIMARY KEY,
        xp INTEGER DEFAULT 0,
        level INTEGER DEFAULT 1,
        coins INTEGER DEFAULT 0,
        warnings INTEGER DEFAULT 0,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
''')

# Settings table
cursor.execute('''
    CREATE TABLE IF NOT EXISTS guild_settings (
        guild_id INTEGER PRIMARY KEY,
        welcome_channel INTEGER,
        log_channel INTEGER,
        auto_role INTEGER,
        prefix TEXT DEFAULT '!'
    )
''')
db.commit()
