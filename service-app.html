React Native

import React, { useState } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet } from 'react-native';

export default function TaskApp() {
  const [tasks, setTasks] = useState([]);

  const addTask = (title) => {
    setTasks([...tasks, { id: Date.now().toString(), title, done: false }]);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>My Tasks</Text>
      <FlatList
        data={tasks}
        keyExtractor={item => item.id}
        renderItem={({ item }) =>
          <TouchableOpacity style={styles.task}>
            <Text>{item.title}</Text>
          </TouchableOpacity>
        }
      />
    </View>
  );
}

Step 3: Backend Integration

Every app needs a backend. I connect your app to cloud services for authentication, data storage, and push notifications.

// Firebase Authentication
import auth from '@react-native-firebase/auth';

const signIn = async (email, password) => {
  const result = await auth().signInWithEmailAndPassword(email, password);
  return result.user;
};

// Firestore Database
import firestore from '@react-native-firebase/firestore';

const saveTask = async (userId, task) => {
  await firestore()
    .collection('users')
    .doc(userId)
    .collection('tasks')
    .add(task);
};
